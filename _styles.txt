   65| function fmtDateTime(dateStr?: string, timeStr?: string): string {
   66|   if (!dateStr) return "";
   67|   try {
   68|     const [y, m, d] = dateStr.split("-").map(Number);
   69|     const datePart = `${m}월 ${d}일`;
   70|     if (!timeStr) return datePart;
   71|     const [h, min] = timeStr.split(":");
   72|     const hNum = parseInt(h, 10);
   73|     const ampm = hNum < 12 ? "오전" : "오후";
   74|     const h12 = hNum === 0 ? 12 : hNum > 12 ? hNum - 12 : hNum;
   75|     return `${datePart} ${ampm} ${h12}시 ${min || "00"}분`;
   76|   } catch { return `${dateStr} ${timeStr || ""}`; }
   77| }
